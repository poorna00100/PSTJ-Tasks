Input: 
nums = [0,1,0,3,12]
Output: 
[1,3,12,0,0]
class Solution {
    public void moveZeroes(int[] nums) {
        int left = 0;
        for(int i = 0; i < nums.length; i++){
        if(nums[i] != 0){
            int temp = nums[left];
            nums[left] = nums[i];
            nums[i] = temp;
            left++;

        }
    }
        
    }
}
