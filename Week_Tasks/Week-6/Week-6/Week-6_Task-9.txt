Input: 
mat = [[1,2,3],[4,5,6],[7,8,9]], k = 1
Output: 
[[12,21,16],[27,45,33],[24,39,28]]
class Solution {
    public int[][] matrixBlockSum(int[][] mat, int k) {
        int m = mat.length;
        int n = mat[0].length;
        int[][] x = new int[m][n];
        for(int i = 0; i < m; i++){
            for(int j = 0; j < n; j++){
                int sum = 0;
                for (int r = Math.max(0, i - k);
                     r <= Math.min(m - 1, i + k); r++) {
                    for (int c = Math.max(0, j - k);
                         c <= Math.min(n - 1, j + k); c++) {
                        sum += mat[r][c];
                    }
                }
                x[i][j] = sum;
            }
        }
        return x;
    }
}
